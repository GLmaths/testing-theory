LTSs that respects the axioms FW-FEEDBACK and BOOMERANG

Class gLtsObaFW (P A : Type) `{gLtsOba P A} :=
  MkgLtsObaFW {
      boomerang p1 η β :
      ∃ p2, non_blocking η -> dual β η
        -> p1 ⟶[ β ] p2 /\ p2 ⟶[ η ] p1 ;
      fwd_feedback {p1 p2 p3 η β } :
      non_blocking η -> dual β η -> p1 ⟶[ η ] p2 -> p2 ⟶[ β ] p3
        -> p1 ⟶⋍ p3 \/ p1 ⋍ p3 ;
    }.
