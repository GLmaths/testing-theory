  Definition must_extensional {P : Type}
    `{!Sts (P * T)} (outcome : T -> Prop)
    (p : P) (t : T) : Prop :=
    forall η : max_exec_from (p, t), exists n fex, mex_take_from n η = Some fex
          /\ outcome (fex_from_last fex).2 .
