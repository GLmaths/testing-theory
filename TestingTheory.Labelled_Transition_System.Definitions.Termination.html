Intensional definition of termination

Inductive terminate_i`{gLts P A} (p : P) : Prop :=
| t_refuses' : p ↛ -> terminate_i p
| tstep' : (exists p', p ⟶ p') -> (∀ q, p ⟶ q -> terminate_i q) -> terminate_i p.
