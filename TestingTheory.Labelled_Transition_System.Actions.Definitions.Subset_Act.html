Definition R `{gLts P A} (p : P) : subset_of A :=
  fun μ => p ⟶[μ].
Definition coR `{gLts P Aproc} `{Ht : ExtAction Atest} `{SA : !SyncAction Aproc Atest}
  (p : P) : subset_of Atest :=
  fun μ1 => exists μ2, p ⟶[μ2] /\ sync μ2 μ1 /\ blocking μ1.
